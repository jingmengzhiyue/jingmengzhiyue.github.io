This article provides complete, runnable Python examples covering binary classification, regression, model persistence and SHAP explanation. The code targets sklearn 1.4+, XGBoost 2.1+, LightGBM 4.0+ and CatBoost 1.2+, and includes comments; it has been run end to end under scikit-learn 1.9.1, XGBoost 3.4.1, LightGBM 4.7.0, CatBoost 1.2.10, shap 0.54.1 and optuna 5.1.0 (Python 3.13).

Version differences to watch

  • XGBoost: early_stopping_rounds, eval_metric and callbacks were deprecated in fit() from 1.6 and removed from fit() in 2.1, so they belong in the constructor (the code here follows that).
  • LightGBM: from 4.0, fit() / train() no longer accept early_stopping_rounds or verbose_eval; use callbacks=[lgb.early_stopping(...)] instead. From 4.7 the sklearn interface gains eval_X / eval_y and marks eval_set deprecated (it still works, with a warning).
  • scikit-learn: AdaBoostClassifier’s SAMME.R was deprecated in 1.4 and removed in 1.6; the algorithm parameter itself was deprecated in 1.6 and removed in 1.8, so passing algorithm=... raises TypeError in new versions.

# ============================================================
# Chapter 11: complete binary classification workflow
# uses sklearn + XGBoost + LightGBM + CatBoost
# ============================================================

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.datasets import make_classification
from sklearn.model_selection import train_test_split, StratifiedKFold, cross_val_score
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import (
    roc_auc_score, roc_curve, precision_recall_curve,
    average_precision_score, classification_report, confusion_matrix
)

# import the boosting models
from sklearn.ensemble import AdaBoostClassifier, GradientBoostingClassifier
from xgboost import XGBClassifier
from lightgbm import LGBMClassifier
from lightgbm import early_stopping as lgb_early_stopping
from lightgbm import log_evaluation as lgb_log_evaluation
from catboost import CatBoostClassifier

# configure fonts for CJK labels (adjust for your system)
plt.rcParams['font.sans-serif'] = ['SimHei', 'DejaVu Sans']
plt.rcParams['axes.unicode_minus'] = False

# ------------------------------------------------------------
# Step 1: generate synthetic classification data
# ------------------------------------------------------------
# 10,000 samples, 30 features, binary target
# n_informative=15: 15 features genuinely related to the label
# n_redundant=10: 10 redundant features, linear combinations of the informative ones
# n_clusters_per_class=2: each class is made of 2 Gaussian clusters, which makes it harder
# flip_y=0.05: 5% of labels flipped, to mimic noise
X, y = make_classification(
    n_samples=10000,           # total samples
    n_features=30,             # total features
    n_informative=15,          # informative features
    n_redundant=10,            # redundant features
    n_clusters_per_class=2,    # clusters per class
    flip_y=0.05,               # label noise rate
    random_state=42            # fixed seed for reproducibility
)

# add a categorical column (mimicking real categorical variables)
# discretize feature 0 into 5 equal-width bins and take the integer codes 0-4
categories = pd.cut(X[:, 0], bins=5, labels=['A', 'B', 'C', 'D', 'E']).codes
X = np.column_stack([X, categories])  # append the categorical column

# record the categorical column index (the last one)
cat_feature_idx = [X.shape[1] - 1]

# note: after np.column_stack the whole array is float, while CatBoost requires
# categorical features to be integers or strings; passing a float array with
# cat_features to CatBoost raises an error. So convert to a DataFrame and cast the
# categorical column to int before handing it to CatBoost.
def to_catboost_frame(X_arr):
    df = pd.DataFrame(X_arr)
    df[cat_feature_idx[0]] = df[cat_feature_idx[0]].astype(int)
    return df

# ------------------------------------------------------------
# Step 2: split and preprocess
# ------------------------------------------------------------
# split 7:1.5:1.5 into train, validation, test
# validation is for early stopping and tuning, test for the final evaluation
X_train_val, X_test, y_train_val, y_test = train_test_split(
    X, y, test_size=0.15, random_state=42, stratify=y
)
X_train, X_val, y_train, y_val = train_test_split(
    X_train_val, y_train_val, test_size=0.176, random_state=42, stratify=y_train_val
)
# final proportions: roughly 70% train, 15% validation, 15% test

# standardize the numeric features (not the categorical one)
# note: tree models don't care about feature scaling, so this isn't required;
# it's only for illustration here, and in practice you can skip it for trees.
scaler = StandardScaler()
X_train_scaled = X_train.copy()
X_val_scaled = X_val.copy()
X_test_scaled = X_test.copy()

# standardize only the numeric features (the first 30 columns)
X_train_scaled[:, :-1] = scaler.fit_transform(X_train[:, :-1])
X_val_scaled[:, :-1] = scaler.transform(X_val[:, :-1])
X_test_scaled[:, :-1] = scaler.transform(X_test[:, :-1])

print(f"Train size: {X_train.shape}")
print(f"Validation size: {X_val.shape}")
print(f"Test size: {X_test.shape}")
print(f"Categorical feature index: {cat_feature_idx}")

# ------------------------------------------------------------
# Step 3: a training function that wraps the early-stopping logic
# ------------------------------------------------------------
def train_and_evaluate_model(model, model_name, X_tr, X_va, y_tr, y_va,
                             X_te, y_te, use_cat_features=False):
    """
    Train and evaluate a model, with early stopping and categorical support.

    Arguments:
        model: the model object
        model_name: name used in log output
        X_tr, X_va, y_tr, y_va: training and validation data
        X_te, y_te: test data
        use_cat_features: whether to pass categorical indices (CatBoost and LightGBM need them)
    """
    print(f"\n{'='*50}")
    print(f"Training: {model_name}")
    print(f"{'='*50}")

    # CatBoost's categorical columns must be integers or strings, so convert first
    if 'CatBoost' in model_name:
        X_tr, X_va, X_te = (to_catboost_frame(a) for a in (X_tr, X_va, X_te))

    # pick the training path by model type
    if use_cat_features and hasattr(model, 'fit'):
        # CatBoost and LightGBM accept a categorical_feature argument
        if 'CatBoost' in model_name:
            model.fit(
                X_tr, y_tr,
                eval_set=(X_va, y_va),           # CatBoost's eval_set format
                cat_features=cat_feature_idx,      # declare categorical features
                early_stopping_rounds=50,          # stop after 50 rounds without improvement
                verbose=False                      # quiet training
            )
        else:  # LightGBM (4.7+ can also use eval_X=..., eval_y=... to avoid the deprecation warning)
            model.fit(
                X_tr, y_tr,
                eval_set=[(X_va, y_va)],
                categorical_feature=cat_feature_idx,
                callbacks=[
                    lgb_early_stopping(50),
                    lgb_log_evaluation(0)
                ]
            )
    else:
        # sklearn models and XGBoost use the standard interface
        if hasattr(model, 'early_stopping_rounds') or 'XGB' in model_name:
            model.fit(
                X_tr, y_tr,
                eval_set=[(X_va, y_va)],
                verbose=False
            )
        else:
            # sklearn models (AdaBoost, GBM) have no early stopping, so just fit
            model.fit(X_tr, y_tr)

    # predicted probabilities (for AUC and the ROC/PR curves)
    if hasattr(model, 'predict_proba'):
        y_proba = model.predict_proba(X_te)[:, 1]
    else:
        # AdaBoost's decision_function returns the weighted vote sum, which needs a
        # sigmoid; in practice sklearn's AdaBoostClassifier does have predict_proba.
        y_proba = model.decision_function(X_te)
        y_proba = 1 / (1 + np.exp(-y_proba))  # sigmoid transform

    # predicted labels (for the classification report)
    y_pred = model.predict(X_te)

    # compute the metrics
    auc = roc_auc_score(y_te, y_proba)
    ap = average_precision_score(y_te, y_proba)  # area under the PR curve

    print(f"Test AUC: {auc:.4f}")
    print(f"Test Average Precision: {ap:.4f}")
    print(f"\nClassification report:\n{classification_report(y_te, y_pred)}")

    return y_proba, auc, model

# ------------------------------------------------------------
# Step 4: train each model
# ------------------------------------------------------------

# 4.1 AdaBoost (baseline)
ada_model = AdaBoostClassifier(
    n_estimators=200,            # 200 weak classifiers
    learning_rate=0.5,           # 0.5 (AdaBoost defaults to 1.0; a little lower is steadier)
    # no algorithm argument: deprecated in sklearn 1.6 and removed in 1.8; SAMME is used internally
    random_state=42
)
ada_proba, ada_auc, ada_trained = train_and_evaluate_model(
    ada_model, "AdaBoost", X_train, X_val, y_train, y_val, X_test, y_test
)

# 4.2 sklearn Gradient Boosting
# note: sklearn 1.4+ recommends HistGradientBoostingClassifier;
# GradientBoostingClassifier is used here to demonstrate classic GBM
gbm_model = GradientBoostingClassifier(
    n_estimators=500,              # 500 trees
    learning_rate=0.1,             # learning rate 0.1
    max_depth=4,                   # depth 4, slightly deeper than the default 3
    subsample=0.8,                 # row sampling 80%
    max_features='sqrt',           # column sampling, sqrt(31) is about 5 features
    random_state=42
)
gbm_proba, gbm_auc, gbm_trained = train_and_evaluate_model(
    gbm_model, "sklearn GBM", X_train, X_val, y_train, y_val, X_test, y_test
)

# 4.3 XGBoost
xgb_model = XGBClassifier(
    n_estimators=1000,             # generously large; early stopping decides the real count
    learning_rate=0.1,             # initial learning rate
    max_depth=5,                   # depth 5
    min_child_weight=3,            # minimum child weight
    subsample=0.8,                 # row sampling
    colsample_bytree=0.8,          # column sampling
    reg_lambda=1,                  # L2 regularization
    reg_alpha=0.1,                 # L1 regularization
    gamma=0.5,                     # minimum split gain
    random_state=42,
    eval_metric='logloss',
    early_stopping_rounds=50       # XGBoost's sklearn interface wants this in the constructor
)
xgb_proba, xgb_auc, xgb_trained = train_and_evaluate_model(
    xgb_model, "XGBoost", X_train, X_val, y_train, y_val, X_test, y_test
)
print(f"XGBoost best iteration: {xgb_trained.best_iteration}")

# 4.4 LightGBM
lgb_model = LGBMClassifier(
    n_estimators=1000,
    learning_rate=0.1,
    num_leaves=31,                 # maximum leaves (roughly depth-5 complexity)
    max_depth=6,                   # bound the depth explicitly to keep leaf-wise growth in check
    feature_fraction=0.8,          # feature sampling (colsample_bytree)
    bagging_fraction=0.8,          # data sampling (subsample)
    bagging_freq=1,                # run bagging every round
    min_data_in_leaf=20,           # minimum samples per leaf (against overfitting)
    lambda_l2=1,                   # L2 regularization
    random_state=42,
    verbose=-1                     # fully quiet
)
lgb_proba, lgb_auc, lgb_trained = train_and_evaluate_model(
    lgb_model, "LightGBM", X_train, X_val, y_train, y_val, X_test, y_test,
    use_cat_features=True
)
print(f"LightGBM best iteration: {lgb_trained.best_iteration_}")

# 4.5 CatBoost
cb_model = CatBoostClassifier(
    iterations=1000,               # iterations
    depth=6,                       # tree depth (symmetric trees)
    learning_rate=0.1,             # learning rate
    l2_leaf_reg=3,                 # L2 regularization on leaves (CatBoost default 3)
    random_strength=1,             # split randomness
    border_count=254,              # bins for numeric features
    loss_function='Logloss',       # log loss
    eval_metric='AUC',             # validation metric
    random_seed=42,
    verbose=False                  # quiet training
)
cb_proba, cb_auc, cb_trained = train_and_evaluate_model(
    cb_model, "CatBoost", X_train, X_val, y_train, y_val, X_test, y_test,
    use_cat_features=True
)
print(f"CatBoost best iteration: {cb_trained.best_iteration_}")

# ------------------------------------------------------------
# Step 5: plot the ROC curves for comparison
# ------------------------------------------------------------
plt.figure(figsize=(10, 8))

# compute each model's ROC curve
fpr_ada, tpr_ada, _ = roc_curve(y_test, ada_proba)
fpr_gbm, tpr_gbm, _ = roc_curve(y_test, gbm_proba)
fpr_xgb, tpr_xgb, _ = roc_curve(y_test, xgb_proba)
fpr_lgb, tpr_lgb, _ = roc_curve(y_test, lgb_proba)
fpr_cb, tpr_cb, _ = roc_curve(y_test, cb_proba)

# draw the curves
plt.plot(fpr_ada, tpr_ada, label=f'AdaBoost (AUC={ada_auc:.3f})', linewidth=2)
plt.plot(fpr_gbm, tpr_gbm, label=f'sklearn GBM (AUC={gbm_auc:.3f})', linewidth=2)
plt.plot(fpr_xgb, tpr_xgb, label=f'XGBoost (AUC={xgb_auc:.3f})', linewidth=2)
plt.plot(fpr_lgb, tpr_lgb, label=f'LightGBM (AUC={lgb_auc:.3f})', linewidth=2)
plt.plot(fpr_cb, tpr_cb, label=f'CatBoost (AUC={cb_auc:.3f})', linewidth=2)

# the diagonal (random guessing)
plt.plot([0, 1], [0, 1], 'k--', label='Random Guess (AUC=0.500)', linewidth=1)

plt.xlabel('False Positive Rate (FPR)', fontsize=12)
plt.ylabel('True Positive Rate (TPR)', fontsize=12)
plt.title('ROC Curve Comparison - Boosting Algorithms', fontsize=14)
plt.legend(loc='lower right', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xlim([0, 1])
plt.ylim([0, 1])
plt.tight_layout()
plt.savefig('roc_comparison.png', dpi=150)
plt.show()

# ------------------------------------------------------------
# Step 6: plot the PR curves for comparison
# ------------------------------------------------------------
plt.figure(figsize=(10, 8))

# compute each model's PR curve
precision_ada, recall_ada, _ = precision_recall_curve(y_test, ada_proba)
precision_gbm, recall_gbm, _ = precision_recall_curve(y_test, gbm_proba)
precision_xgb, recall_xgb, _ = precision_recall_curve(y_test, xgb_proba)
precision_lgb, recall_lgb, _ = precision_recall_curve(y_test, lgb_proba)
precision_cb, recall_cb, _ = precision_recall_curve(y_test, cb_proba)

plt.plot(recall_ada, precision_ada, label=f'AdaBoost (AP={average_precision_score(y_test, ada_proba):.3f})', linewidth=2)
plt.plot(recall_gbm, precision_gbm, label=f'sklearn GBM (AP={average_precision_score(y_test, gbm_proba):.3f})', linewidth=2)
plt.plot(recall_xgb, precision_xgb, label=f'XGBoost (AP={average_precision_score(y_test, xgb_proba):.3f})', linewidth=2)
plt.plot(recall_lgb, precision_lgb, label=f'LightGBM (AP={average_precision_score(y_test, lgb_proba):.3f})', linewidth=2)
plt.plot(recall_cb, precision_cb, label=f'CatBoost (AP={average_precision_score(y_test, cb_proba):.3f})', linewidth=2)

plt.xlabel('Recall', fontsize=12)
plt.ylabel('Precision', fontsize=12)
plt.title('Precision-Recall Curve Comparison', fontsize=14)
plt.legend(loc='lower left', fontsize=10)
plt.grid(True, alpha=0.3)
plt.xlim([0, 1])
plt.ylim([0, 1])
plt.tight_layout()
plt.savefig('pr_comparison.png', dpi=150)
plt.show()

# ------------------------------------------------------------
# Step 7: feature importance (XGBoost as the example)
# ------------------------------------------------------------
plt.figure(figsize=(12, 8))

# XGBoost feature importance (gain based)
feature_importance = xgb_trained.feature_importances_
feature_names = [f'feature_{i}' for i in range(X.shape[1])]
feature_names[-1] = 'category_feature'  # the last column is categorical

# sort by importance
idx = np.argsort(feature_importance)[::-1][:15]  # top 15

plt.barh(range(len(idx)), feature_importance[idx], align='center')
plt.yticks(range(len(idx)), [feature_names[i] for i in idx])
plt.xlabel('Feature Importance (Gain)', fontsize=12)
plt.ylabel('Feature', fontsize=12)
plt.title('XGBoost Top 15 Feature Importances', fontsize=14)
plt.gca().invert_yaxis()  # highest at the top
plt.tight_layout()
plt.savefig('feature_importance.png', dpi=150)
plt.show()

# print the AUC summary
print("\n" + "="*60)
print("Test AUC by model:")
print(f"AdaBoost:    {ada_auc:.4f}")
print(f"sklearn GBM: {gbm_auc:.4f}")
print(f"XGBoost:     {xgb_auc:.4f}")
print(f"LightGBM:    {lgb_auc:.4f}")
print(f"CatBoost:    {cb_auc:.4f}")
print("="*60)

# Chapter 11: complete regression workflow
# comparing XGBoost + LightGBM + CatBoost
# ============================================================

import numpy as np
import matplotlib.pyplot as plt
from sklearn.datasets import make_regression
from sklearn.model_selection import train_test_split, cross_val_score
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score

from sklearn.ensemble import AdaBoostRegressor, GradientBoostingRegressor
from xgboost import XGBRegressor
from lightgbm import LGBMRegressor
from lightgbm import early_stopping as lgb_early_stopping
from lightgbm import log_evaluation as lgb_log_evaluation
from catboost import CatBoostRegressor

# ------------------------------------------------------------
# Step 1: generate regression data
# ------------------------------------------------------------
# 5,000 samples, 20 features
# n_informative=10: 10 informative features
# noise=10.0: Gaussian noise with standard deviation 10, to make prediction harder
X, y = make_regression(
    n_samples=5000,
    n_features=20,
    n_informative=10,
    noise=10.0,                    # noise level
    random_state=42
)

# split 8:2 into train and test

    X, y, test_size=0.2, random_state=42
)

print(f"Test size: {X_test.shape}")

# ------------------------------------------------------------
# Step 2: a training and evaluation function for regressors
# ------------------------------------------------------------
def train_and_evaluate_regressor(model, model_name, X_tr, X_te, y_tr, y_te):
    """
    Train and evaluate a regression model, returning predictions and metrics.
    """
    print(f"\n{'='*50}")
    print(f"Training regressor: {model_name}")
    print(f"{'='*50}")

    # XGBoost / LightGBM / CatBoost early stop on a separate validation set;
    # the sklearn models here don't early stop.
    # note: the model name is "LightGBM Regressor", which doesn't contain 'LGBM',
    # so the check must be on 'LightGBM'; otherwise LightGBM falls into the else
    # branch and trains the full 1000 rounds without early stopping.
    if 'XGB' in model_name or 'LightGBM' in model_name or 'CatBoost' in model_name:
        # carve a validation set out of the training data for early stopping
        X_tr_inner, X_val_inner, y_tr_inner, y_val_inner = train_test_split(
            X_tr, y_tr, test_size=0.1, random_state=42
        )
        if 'LightGBM' in model_name:
            model.fit(
                X_tr_inner, y_tr_inner,
                eval_set=[(X_val_inner, y_val_inner)],
                callbacks=[
                    lgb_early_stopping(30),
                    lgb_log_evaluation(0)
                ]
            )
        elif 'CatBoost' in model_name:
            model.fit(
                X_tr_inner, y_tr_inner,
                eval_set=(X_val_inner, y_val_inner),
                early_stopping_rounds=30,
                verbose=False
            )
        else:  # XGBoost: early_stopping_rounds lives in the constructor
            model.fit(
                X_tr_inner, y_tr_inner,
                eval_set=[(X_val_inner, y_val_inner)],
                verbose=False
            )
    else:
        model.fit(X_tr, y_tr)

    # predict
    y_pred = model.predict(X_te)

    # compute the metrics
    mse = mean_squared_error(y_te, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_te, y_pred)
    r2 = r2_score(y_te, y_pred)

    print(f"MSE:  {mse:.4f}")
    print(f"RMSE: {rmse:.4f}")
    print(f"MAE:  {mae:.4f}")
    print(f"R^2:  {r2:.4f}")

    return y_pred, {'mse': mse, 'rmse': rmse, 'mae': mae, 'r2': r2}

11.5 Feature importance: types and differences

Step 7 of 11.1 drew an XGBoost importance bar chart from feature_importances_. Note that “feature importance” means different things in different libraries, and switching the importance type can reorder the features substantially.

Library / methodTypeMeaningDefault
XGBoostweightNumber of times the feature was used to split, across all trees—
XGBoostgain / total_gainAverage / total gain of that feature’s splitsfeature_importances_ for tree models in the sklearn interface uses gain
XGBoostcover / total_coverAverage / total Hessian-weighted sample coverage of that feature’s splits—
LightGBMsplit / gainTimes used to split / total gain contributedthe sklearn interface defaults to importance_type='split'
CatBoostPredictionValuesChange and othersSee the code in CatBoostPredictionValuesChange
sklearnpermutation importanceDrop in the metric after shuffling one column on held-out datamodel agnostic

A few things to watch:

  • Split- and gain-based importance is computed on the training data and doesn’t directly reflect a feature’s contribution to generalization; both scikit-learn’s docs and Strobl et al. (2007) note that this kind of importance favors features with many distinct values (high cardinality, continuous).
  • Different types can rank features differently: weight/split only count uses, which often puts features with many small splits on top, while gain is closer to “contribution to reducing the loss.” Lundberg et al. (2020) also point out consistency problems with these global measures and recommend the mean absolute SHAP value $\frac{1}{n}\sum_{i=1}^{n}\lvert\phi_j(x_i)\rvert$ as a global importance measure (see 11.4).
  • Compute permutation importance on a validation or test set. When two features are highly correlated, shuffling one still leaves the model access to the information through the other, and both importances are underestimated.

# Section 11.5: feature importance types and differences
# reusing the models and data from 11.1 (xgb_trained, lgb_trained, X_val and so on)
# ============================================================
from sklearn.inspection import permutation_importance

# 1. XGBoost: Booster.get_score supports 5 importance types
#    keys are feature names; numpy input defaults to 'f0', 'f1', ...; features never
#    used to split don't appear in the result
booster = xgb_trained.get_booster()
for imp_type in ['weight', 'gain', 'cover', 'total_gain', 'total_cover']:
    score = booster.get_score(importance_type=imp_type)
    top3 = sorted(score.items(), key=lambda kv: kv[1], reverse=True)[:3]
    print(f"XGBoost {imp_type:<12}: {[(k, round(v, 2)) for k, v in top3]}")

# 2. LightGBM: split (split count, the sklearn default) and gain (total gain)
lgb_split = lgb_trained.booster_.feature_importance(importance_type='split')
lgb_gain = lgb_trained.booster_.feature_importance(importance_type='gain')
print("LightGBM split top 3 features:", [feature_names[i] for i in np.argsort(lgb_split)[::-1][:3]])
print("LightGBM gain  top 3 features:", [feature_names[i] for i in np.argsort(lgb_gain)[::-1][:3]])

# 3. permutation importance: shuffle each column on the validation set and see how far
#    AUC drops (works for any model)
perm = permutation_importance(
    xgb_trained, X_val, y_val,
    scoring='roc_auc', n_repeats=5, random_state=42, n_jobs=-1
)
for i in np.argsort(perm.importances_mean)[::-1][:5]:
    print(f"{feature_names[i]:<18} AUC drop {perm.importances_mean[i]:.4f} ± {perm.importances_std[i]:.4f}")
